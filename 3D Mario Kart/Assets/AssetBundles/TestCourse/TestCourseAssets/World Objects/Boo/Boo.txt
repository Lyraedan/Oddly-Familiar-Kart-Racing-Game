local random_target = require("random_target")

local origin = nil
local target = nil
local waitTime = 0

local speed = 25
local radius = 25
local height = 10

local function Start(self)
    self.actor.name = "Boo"

    origin = self.actor.transform.position
    target = random_target.get(origin, radius, height)
end

local function Update(self, dt)
    if waitTime > 0 then
        waitTime = waitTime - dt
        self.animator:SetBool("Idle", true)
        return
    end

    local pos = self.actor.transform.position
    local dist = vector3:Distance(pos, target)

    if dist < 0.2 then
        waitTime = 1 + math.random() * 2
        target = random_target.get(origin, radius, height)
        self.animator:SetBool("Idle", true)
        return
    end

    self.actor.transform:LookAt(vector3:New(target.x, pos.y, target.z))
    self.animator:SetBool("Idle", false)

    self.actor.transform:MoveTowards(target, speed * dt)
end

return {
    Start = Start,
    Update = Update,
}
