local origin = nil
local target = nil
local waitTime = 0

local speed = 25
local radius = 25
local height = 10

local function RandomTarget()
    if math.random() < 0.25 then
        return { x = origin.x, y = origin.y, z = origin.z }
    end

    local rx = (math.random() * 2 - 1) * radius
    local ry = (math.random() * 2 - 1) * height
    local rz = (math.random() * 2 - 1) * radius

    return {
        x = origin.x + rx,
        y = origin.y + ry,
        z = origin.z + rz
    }
end

local function Start(self)
    self.actor.name = "Boo"

    local pos = self.actor.transform.position
    origin = { x = pos.x, y = pos.y, z = pos.z }

    target = RandomTarget()
end

local function Update(self, dt)

	-- We're waiting
    if waitTime > 0 then
        waitTime = waitTime - dt
        self.animator:SetBool("Idle", true)
        return
    end

    local pos = self.actor.transform.position

    local dx = target.x - pos.x
    local dy = target.y - pos.y
    local dz = target.z - pos.z

    local dist = math.sqrt(dx*dx + dy*dy + dz*dz)

    -- Reached target
    if dist < 0.2 then
        waitTime = 1 + math.random() * 2
        target = RandomTarget()
        self.animator:SetBool("Idle", true)
        return
    end

    self.actor.transform:LookAt(target.x, pos.y, target.z)
    self.animator:SetBool("Idle", false)

    self.actor.transform:MoveTowards(
        target.x,
        target.y,
        target.z,
        speed * dt
    )
end

return {
    Start = Start,
    Update = Update,
}
