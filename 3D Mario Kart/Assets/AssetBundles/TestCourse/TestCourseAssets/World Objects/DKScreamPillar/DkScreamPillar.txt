function PlayRandomAnimation(self)
    local roll = math.random(1, 100)
	if roll <= 10 then
		-- Remain Idle (10%)
    elseif roll <= 30 then
        Drop(self) -- 20%
    elseif roll <= 80 then
        Swing(self) -- 50%
    else
        DropAndSwing(self) -- 20%
    end
end

function Drop(self)
    local dropLength = math.random(1, 2)
    self.animator:SetBool("Drop", true)
    wait(dropLength)
    self.animator:SetBool("Drop", false)
end

function Swing(self)
    local swingLength = math.random(2, 3)
    self.animator:SetBool("Swing", true)
    wait(swingLength)
    self.animator:SetBool("Swing", false)
end

function DropAndSwing(self)
	local length = math.random(3, 4)
    self.animator:SetBool("Swing", true)
	self.animator:SetBool("Drop", true)
    wait(length)
    self.animator:SetBool("Swing", false)
	self.animator:SetBool("Drop", false)
end

function animator_PlayDropSound(self)
	--self.drop:Play()
	print("Drop sound")
end

function animator_PlaySwingSound(self)
	--self.shake:Play()
	print("Swing sound")
end

local function Start(self)
    self.actor.name = "Chandelier"

    startCoroutine(function()
        while true do
            PlayRandomAnimation(self)
            wait(1) -- Wait a second before playing the next anim
        end
    end)
end

return {
    Start = Start,
	PlayDropSound = animator_PlayDropSound, -- Triggered via the animation
	PlayShakeSound = animator_PlaySwingSound -- Triggered via the animation
}
