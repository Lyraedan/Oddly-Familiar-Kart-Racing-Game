function PlayRandomAnimation(self)
    if anim == 1 then
		-- Remain Idle
    elseif anim == 2 then
        Drop(self)
    elseif anim == 3 then
        Shake(self)
    elseif anim == 4 then
        DropAndShake(self)
    end
end

function Drop(self)
    local dropLength = math.random(1, 2)
    self.animator:SetBool("Drop", true)
    wait(dropLength)
    self.animator:SetBool("Drop", false)
end

function Shake(self)
    local shakeLength = math.random(2, 3)
    self.animator:SetBool("Shake", true)
    wait(shakeLength)
    self.animator:SetBool("Shake", false)
end

function DropAndShake(self)
	local length = math.random(3, 4)
    self.animator:SetBool("Shake", true)
	self.animator:SetBool("Drop", true)
    wait(length)
    self.animator:SetBool("Shake", false)
	self.animator:SetBool("Drop", false)
end

function animator_PlayDropSound(self)
	--self.drop:Play()
	print("Drop sound")
end

function animator_PlayShakeSound(self)
	--self.shake:Play()
	print("Shake sound")
end

local function Start(self)
    self.actor.name = "DkScarePillar"

    startCoroutine(function()
        while true do
            PlayRandomAnimation(self)
            wait(1) -- Wait a second before playing the next anim
        end
    end)
end

return {
    Start = Start,
	PlayDropSound = animator_PlayDropSound, -- Triggered via the animation
	PlayShakeSound = animator_PlayShakeSound -- Triggered via the animation
}
