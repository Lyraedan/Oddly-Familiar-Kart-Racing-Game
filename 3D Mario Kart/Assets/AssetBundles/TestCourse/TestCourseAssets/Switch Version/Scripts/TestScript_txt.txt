return {

Start = function(self)

    print("Lantern started")

    -- rename objects
    self.actor:SetName("Lantern")
    self.flame.actor:SetName("Flame Light")

    -- turn flame light on
    self.flame:SetEnabled(true)
    self.flame:SetIntensity(2.0)

    -- start coroutine example
    startCoroutine(function()
        print("Coroutine")
        while true do
            wait(1)  -- yields for 1 second
            print("Coroutine tick")
        end
    end)

end,

Update = function(self, dt)

    -- move and rotate the Lantern
    self.actor.transform:MoveForward(2 * dt)
    self.actor.transform:Rotate(0, 30 * dt, 0)

    -- flicker the flame
    local flicker = 1.8 + math.random() * 0.4 * 100
    self.flame:SetIntensity(flicker)
    self.flame.SetColor(1.0, 0.5 + math.random() * 0.5, 0.0)

end

}
