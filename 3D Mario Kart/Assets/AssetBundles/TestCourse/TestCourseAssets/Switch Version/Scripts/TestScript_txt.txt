function PrintTransform(obj)
    if obj == nil or obj.transform == nil then
        print("Object or transform is nil")
        return
    end

    local t = obj.transform

    local pos = t.position
    local rot = t.rotation
    local scale = t.scale

    print("Transform of " .. (obj.name or "unnamed object") .. ":")
    print(string.format("  Position:  x=%.2f, y=%.2f, z=%.2f", pos.x, pos.y, pos.z))
    print(string.format("  Rotation:  x=%.2f, y=%.2f, z=%.2f", rot.x, rot.y, rot.z))
    print(string.format("  Scale:     x=%.2f, y=%.2f, z=%.2f", scale.x, scale.y, scale.z))
end

local function Start(self)
    print("Lantern started")

    -- Rename objects
    self.actor.name = "Lantern"
    self.flame.actor.name = "Flame Light"
    self.flame.actor.transform:SetParent(self.fire.actor) -- Parent the fire audio to the flame light

    -- Turn flame light on
    self.flame:SetEnabled(true)
    self.flame:SetIntensity(2.0)

    -- Start coroutine example
    startCoroutine(function()
        print("Coroutine")
        while true do
            wait(1) -- Yield for 1 second
            print("Coroutine tick")
        end
    end)

    TestMathf();
    PrintTransform(self.actor)
end

local function Update(self, dt)
    -- Move and rotate the lantern
    self.actor.transform:MoveForward(2 * dt)
    self.actor.transform:Rotate(0, 30 * dt, 0)

    -- Flicker the flame
    local flicker = 1.8 + math.random() * 0.4
    self.flame:SetIntensity(flicker)
    self.flame:SetColor(
        1.0,
        0.5 + math.random() * 0.5,
        0.0
    )
end

return {
    Start = Start,
    Update = Update
}
